cmake_minimum_required(VERSION 3.20)
project(MeshSculpt VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
include(GNUInstallDirs)

option(MESH_SCULPT_BUILD_APP "Build the Mesh Sculpt desktop application" ON)
option(MESH_SCULPT_BUILD_TESTS "Build Mesh Sculpt tests" ${BUILD_TESTING})
option(MESH_SCULPT_WARNINGS_AS_ERRORS "Treat project warnings as errors" OFF)
option(MESH_SCULPT_ENABLE_SANITIZERS "Enable supported sanitizers" OFF)
option(MESH_SCULPT_INSTALL "Enable install rules" ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(cmake/Dependencies.cmake)

add_library(mesh_sculpt_warnings INTERFACE)
if(MSVC)
    target_compile_options(mesh_sculpt_warnings INTERFACE /W4 /permissive-)
    if(MESH_SCULPT_WARNINGS_AS_ERRORS)
        target_compile_options(mesh_sculpt_warnings INTERFACE /WX)
    endif()
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(mesh_sculpt_warnings INTERFACE -Wall -Wextra -Wpedantic)
    if(MESH_SCULPT_WARNINGS_AS_ERRORS)
        target_compile_options(mesh_sculpt_warnings INTERFACE -Werror)
    endif()
endif()

add_library(mesh_sculpt_sanitizers INTERFACE)
if(MESH_SCULPT_ENABLE_SANITIZERS)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
        target_compile_options(mesh_sculpt_sanitizers INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(mesh_sculpt_sanitizers INTERFACE -fsanitize=address,undefined)
    else()
        message(WARNING "Mesh Sculpt sanitizers are supported only by GCC/Clang on non-Windows platforms")
    endif()
endif()

set(mesh_sculpt_first_party mesh_sculpt_warnings mesh_sculpt_sanitizers)
add_library(mesh_sculpt_core
    src/sculpt/SculptMesh.cpp src/sculpt/MeshDocument.cpp src/sculpt/MeshSelection.cpp
    src/sculpt/MeshPicker.cpp src/sculpt/VertexDragManipulator.cpp
    src/sculpt/MeshSculptRenderData.cpp src/sculpt/MeshEditorState.cpp src/sculpt/MeshOverlayData.cpp)
target_include_directories(mesh_sculpt_core PUBLIC ${PROJECT_SOURCE_DIR}/include)
target_link_libraries(mesh_sculpt_core PUBLIC glm::glm PRIVATE ${mesh_sculpt_first_party})

add_library(mesh_sculpt_io src/io/GeometryFileFormat.cpp src/io/GeometryFileService.cpp)
target_include_directories(mesh_sculpt_io PUBLIC ${PROJECT_SOURCE_DIR}/include)
target_link_libraries(mesh_sculpt_io PUBLIC mesh_sculpt_core PRIVATE nlohmann_json::nlohmann_json ${mesh_sculpt_first_party})

add_library(mesh_sculpt_platform src/platform/AssetLocator.cpp)
target_include_directories(mesh_sculpt_platform PUBLIC ${PROJECT_SOURCE_DIR}/include)
target_compile_definitions(mesh_sculpt_platform PRIVATE MESH_SCULPT_DEVELOPMENT_ASSET_ROOT="${PROJECT_SOURCE_DIR}/assets")
target_link_libraries(mesh_sculpt_platform PRIVATE ${mesh_sculpt_first_party})

if(MESH_SCULPT_BUILD_APP)
    find_package(OpenGL REQUIRED)
    mesh_sculpt_enable_app_dependencies()
    add_executable(mesh_sculpt src/main.cpp src/app/SculptApplication.cpp src/app/MeshSculptController.cpp
        src/platform/Window.cpp src/render/DynamicMesh.cpp src/render/FPSCamera.cpp src/render/Shader.cpp
        src/sculpt/MeshSculptTool.cpp src/sculpt/MeshSculptRenderer.cpp src/sculpt/MeshSculptUi.cpp
        src/sculpt/MeshSculptEditorUi.cpp)
    target_include_directories(mesh_sculpt PRIVATE ${PROJECT_SOURCE_DIR}/include)
    target_compile_definitions(mesh_sculpt PRIVATE GLM_FORCE_RADIANS)
    target_link_libraries(mesh_sculpt PRIVATE mesh_sculpt_core mesh_sculpt_io mesh_sculpt_platform glad glfw
        glm::glm mesh_sculpt_imgui portable_file_dialogs OpenGL::GL ${mesh_sculpt_first_party})
    add_custom_command(TARGET mesh_sculpt POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${PROJECT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:mesh_sculpt>/assets)
endif()

if(BUILD_TESTING AND MESH_SCULPT_BUILD_TESTS)
    function(mesh_sculpt_test target source library)
        add_executable(${target} ${source})
        target_link_libraries(${target} PRIVATE ${library} ${mesh_sculpt_first_party})
        add_test(NAME ${target} COMMAND ${target})
    endfunction()
    mesh_sculpt_test(sculpt_mesh_tests tests/SculptMeshTests.cpp mesh_sculpt_core)
    mesh_sculpt_test(mesh_selection_tests tests/MeshSelectionTests.cpp mesh_sculpt_core)
    mesh_sculpt_test(mesh_document_tests tests/MeshDocumentTests.cpp mesh_sculpt_core)
    mesh_sculpt_test(geometry_file_format_tests tests/GeometryFileFormatTests.cpp mesh_sculpt_io)
    mesh_sculpt_test(geometry_file_service_tests tests/GeometryFileServiceTests.cpp mesh_sculpt_io)
    mesh_sculpt_test(mesh_picker_tests tests/MeshPickerTests.cpp mesh_sculpt_core)
    mesh_sculpt_test(vertex_drag_manipulator_tests tests/VertexDragManipulatorTests.cpp mesh_sculpt_core)
    mesh_sculpt_test(mesh_sculpt_render_data_tests tests/MeshSculptRenderDataTests.cpp mesh_sculpt_core)
    mesh_sculpt_test(mesh_editor_state_tests tests/MeshEditorTextStateTests.cpp mesh_sculpt_core)
    mesh_sculpt_test(mesh_overlay_data_tests tests/MeshOverlayDataTests.cpp mesh_sculpt_core)
    mesh_sculpt_test(asset_locator_tests tests/AssetLocatorTests.cpp mesh_sculpt_platform)
    target_compile_definitions(geometry_file_format_tests PRIVATE MESH_SCULPT_EXAMPLE_FILE="${PROJECT_SOURCE_DIR}/examples/cube.meshgeo")
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME standalone_independence COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/scripts/check_standalone_independence.py)
    endif()
endif()

find_program(CLANG_FORMAT_EXECUTABLE NAMES clang-format)
if(CLANG_FORMAT_EXECUTABLE)
    file(GLOB_RECURSE format_files CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/src/*.cpp ${PROJECT_SOURCE_DIR}/include/*.h ${PROJECT_SOURCE_DIR}/tests/*.cpp)
    add_custom_target(format COMMAND ${CLANG_FORMAT_EXECUTABLE} -i ${format_files})
    add_custom_target(format-check COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/scripts/check_format.py --clang-format ${CLANG_FORMAT_EXECUTABLE})
endif()

if(MESH_SCULPT_INSTALL)
    if(TARGET mesh_sculpt)
        install(TARGETS mesh_sculpt RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    endif()
    install(DIRECTORY assets/shaders DESTINATION ${CMAKE_INSTALL_DATADIR}/mesh-sculpt)
    install(FILES README.md LICENSE_STATUS.md THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
endif()

set(CPACK_GENERATOR "TGZ;ZIP")
set(CPACK_PACKAGE_NAME MeshSculpt)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "MeshSculpt-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}")
include(CPack)
